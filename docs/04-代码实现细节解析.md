```go
func main() {
    // ========== 1. 命令行参数解析 ==========
    verbose := flag.Bool("verbose", false, "enable verbose logging")
    flag.Parse()
    
    // 为什么用 flag？
    // - 支持命令行参数: go run chat.go --verbose
    // - 自动生成帮助: go run chat.go --help
    
    // ========== 2. 日志配置 ==========
    if *verbose {
        // 详细模式：输出到 stderr，包含文件名和行号
        log.SetOutput(os.Stderr)
        log.SetFlags(log.LstdFlags | log.Lshortfile)
        log.Println("Verbose logging enabled")
    } else {
        // 普通模式：输出到 stdout，不包含额外信息
        log.SetOutput(os.Stdout)
        log.SetFlags(0)
        log.SetPrefix("")
    }
    
    // 为什么区分 stderr 和 stdout？
    // - stdout: 程序的主要输出（可以重定向到文件）
    // - stderr: 错误和日志信息（不受重定向影响）
    
    // ========== 3. 创建客户端 ==========
    client := anthropic.NewClient()
    // 自动读取 ANTHROPIC_API_KEY 环境变量
    // 等价于: anthropic.NewClient(os.Getenv("ANTHROPIC_API_KEY"))
    
    if *verbose {
        log.Println("Anthropic client initialized")
    }
    
    // ========== 4. 创建用户输入函数 ==========
    scanner := bufio.NewScanner(os.Stdin)
    getUserMessage := func() (string, bool) {
        if !scanner.Scan() {
            // Scan() 返回 false 表示:
            // - EOF (Ctrl+D)
            // - 读取错误
            return "", false
        }
        return scanner.Text(), true
    }
    
    // 为什么用函数而不是直接读取？
    // 1. 依赖注入：测试时可以替换为 mock 函数
    // 2. 封装：隐藏 scanner 的细节
    // 3. 复用：可以在不同地方调用
    
    // ========== 5. 创建并运行 Agent ==========
    agent := NewAgent(&client, getUserMessage, *verbose)
    err := agent.Run(context.TODO())
    
    if err != nil {
        fmt.Printf("Error: %s\n", err.Error())
    }
}
```

#### Agent 结构体

```go
func NewAgent(client *anthropic.Client, 
    getUserMessage func() (string, bool), 
    verbose bool) *Agent {
    
    return &Agent{
        client:         client,
        getUserMessage: getUserMessage,
        verbose:        verbose,
    }
}

type Agent struct {
    client         *anthropic.Client        // Claude API 客户端
    getUserMessage func() (string, bool)    // 获取用户输入的函数
    verbose        bool                     // 是否详细日志
}

// 为什么这样设计？
// 1. client: 需要跨多次调用复用（保持连接）
// 2. getUserMessage: 函数式设计，方便测试
// 3. verbose: 调试开关，影响日志输出
```

#### Run 方法（事件循环）

```go
func (a *Agent) Run(ctx context.Context) error {
    // ========== 1. 初始化对话历史 ==========
    conversation := []anthropic.MessageParam{}
    // 为什么用切片？
    // - 动态大小：对话历史会不断增长
    // - 有序：保持对话顺序
    // - API 要求：Anthropic API 期望消息数组
    
    if a.verbose {
        log.Println("Starting chat session")
    }
    fmt.Println("Chat with Claude (use 'ctrl-c' to quit)")
    
    // ========== 2. 主循环 ==========
    for {
        // 2.1 获取用户输入
        fmt.Print("\u001b[94mYou\u001b[0m: ")
        // \u001b[94m: ANSI 蓝色
        // \u001b[0m: 重置颜色
        
        userInput, ok := a.getUserMessage()
        if !ok {
            // 用户输入结束（EOF 或错误）
            if a.verbose {
                log.Println("User input ended, breaking from chat loop")
            }
            break
        }
        
        // 2.2 跳过空输入
        if userInput == "" {
            if a.verbose {
                log.Println("Skipping empty message")
            }
            continue
        }
        
        if a.verbose {
            log.Printf("User input received: %q", userInput)
        }
        
        // 2.3 构造用户消息
        userMessage := anthropic.NewUserMessage(
            anthropic.NewTextBlock(userInput)
        )
        // 等价的 JSON:
        // {
        //   "role": "user",
        //   "content": [
        //     {"type": "text", "text": "userInput 的内容"}
        //   ]
        // }
        
        conversation = append(conversation, userMessage)
        
        // 2.4 调用 Claude
        if a.verbose {
            log.Printf("Sending message to Claude, conversation length: %d", 
                len(conversation))
        }
        
        message, err := a.runInference(ctx, conversation)
        if err != nil {
            if a.verbose {
                log.Printf("Error during inference: %v", err)
            }
            return err
        }
        
        // 2.5 保存 Claude 的回复到历史
        conversation = append(conversation, message.ToParam())
        // ToParam() 将 *anthropic.Message 转换为 MessageParam
        // 这样下一轮对话时 Claude 能看到自己的回复
        
        if a.verbose {
            log.Printf("Received response from Claude with %d content blocks", 
                len(message.Content))
        }
        
        // 2.6 处理响应内容
        for _, content := range message.Content {
            switch content.Type {
            case "text":
                fmt.Printf("\u001b[93mClaude\u001b[0m: %s\n", content.Text)
                // \u001b[93m: ANSI 黄色
            }
            // chat.go 只处理 "text" 类型
            // read.go 等后续版本会处理 "tool_use" 类型
        }
    }
    
    if a.verbose {
        log.Println("Chat session ended")
    }
    return nil
}
```

```go
func (a *Agent) runInference(ctx context.Context, 
    conversation []anthropic.MessageParam) (*anthropic.Message, error) {
    
    if a.verbose {
        log.Printf("Making API call to Claude with model: %s", 
            anthropic.ModelClaude3_7SonnetLatest)
    }
    
    // ========== 核心 API 调用 ==========
    message, err := a.client.Messages.New(ctx, anthropic.MessageNewParams{
        Model:     anthropic.ModelClaude3_7SonnetLatest,
        // 可用模型:
        // - claude-3-5-sonnet-20241022 (推荐，性价比高)
        // - claude-3-5-haiku-20241022 (最快，最便宜)
        // - claude-3-opus-20240229 (最强，最贵)
        
        MaxTokens: int64(1024),
        // 限制最大生成 token 数
        // - 1 token ≈ 4 个英文字符 或 1 个汉字
        // - 1024 tokens ≈ 750 个英文单词
        // - 可以调大（如 4096）获得更长回复
        
        Messages: conversation,
        // 完整的对话历史
        // Claude 会看到所有历史消息
    })
    
    if a.verbose {
        if err != nil {
            log.Printf("API call failed: %v", err)
        } else {
            log.Printf("API call successful, response received")
        }
    }
    
    return message, err
}
```

```go
type ReadFileInput struct {
    Path string `json:"path" jsonschema_description:"The relative path of a file in the working directory."`
}

// 2. 生成 Schema
var ReadFileInputSchema = GenerateSchema[ReadFileInput]()

// 3. 泛型生成函数
func GenerateSchema[T any]() anthropic.ToolInputSchemaParam {
    reflector := jsonschema.Reflector{
        AllowAdditionalProperties: false,  // 不允许额外字段
        DoNotReference:            true,   // 内联所有定义
    }
    var v T  // 创建类型 T 的零值
    
    schema := reflector.Reflect(v)
    // 通过反射分析结构体字段和标签
    
    return anthropic.ToolInputSchemaParam{
        Properties: schema.Properties,
    }
}

// 生成的 Schema 大致等价于:
// {
//   "type": "object",
//   "properties": {
//     "path": {
//       "type": "string",
//       "description": "The relative path of a file..."
//     }
//   },
//   "required": ["path"]
// }
```

```go
func ReadFile(input json.RawMessage) (string, error) {
    // ========== 1. 解析输入 ==========
    readFileInput := ReadFileInput{}
    err := json.Unmarshal(input, &readFileInput)
    // json.Unmarshal: 将 JSON 字节转换为 Go 结构体
    
    if err != nil {
        panic(err)  // 参数格式错误，这是程序 bug，直接 panic
    }
    // 为什么 panic 而不是返回 error？
    // - 这不是运行时错误，而是代码错误
    // - Claude 应该总是发送正确格式的参数
    
    // ========== 2. 记录日志 ==========
    log.Printf("Reading file: %s", readFileInput.Path)
    
    // ========== 3. 读取文件 ==========
    content, err := os.ReadFile(readFileInput.Path)
    // os.ReadFile: 读取整个文件到内存
    // 返回 []byte
    
    if err != nil {
        log.Printf("Failed to read file %s: %v", readFileInput.Path, err)
        return "", err  // 文件不存在或权限错误，返回 error
    }
    
    // ========== 4. 返回结果 ==========
    log.Printf("Successfully read file %s (%d bytes)", 
        readFileInput.Path, len(content))
    return string(content), nil
    // string(content): 将 []byte 转换为 string
}
```

```go
func (a *Agent) Run(ctx context.Context) error {
    conversation := []anthropic.MessageParam{}
    // ...
    
    for {
        // ... 获取用户输入 ...
        
        message, err := a.runInference(ctx, conversation)
        // ...
        
        conversation = append(conversation, message.ToParam())
        
        // ========== 新增：工具调用循环 ==========
        for {
            var toolResults []anthropic.ContentBlockParamUnion
            var hasToolUse bool
            
            for _, content := range message.Content {
                switch content.Type {
                case "text":
                    // 显示文本回复
                    fmt.Printf("\u001b[93mClaude\u001b[0m: %s\n", content.Text)
                    
                case "tool_use":
                    // ========== 处理工具调用 ==========
                    hasToolUse = true
                    toolUse := content.AsToolUse()
                    // AsToolUse() 提取工具调用信息
                    
                    if a.verbose {
                        log.Printf("Tool use detected: %s with input: %s", 
                            toolUse.Name, string(toolUse.Input))
                    }
                    
                    fmt.Printf("\u001b[96mtool\u001b[0m: %s(%s)\n", 
                        toolUse.Name, string(toolUse.Input))
                    // \u001b[96m: ANSI 青色
                    
                    // ========== 查找并执行工具 ==========
                    var toolResult string
                    var toolError error
                    var toolFound bool
                    
                    for _, tool := range a.tools {
                        if tool.Name == toolUse.Name {
                            if a.verbose {
                                log.Printf("Executing tool: %s", tool.Name)
                            }
                            
                            // 执行工具函数
                            toolResult, toolError = tool.Function(toolUse.Input)
                            
                            fmt.Printf("\u001b[92mresult\u001b[0m: %s\n", toolResult)
                            // \u001b[92m: ANSI 绿色
                            
                            if toolError != nil {
                                fmt.Printf("\u001b[91merror\u001b[0m: %s\n", 
                                    toolError.Error())
                                // \u001b[91m: ANSI 红色
                            }
                            
                            toolFound = true
                            break
                        }
                    }
                    
                    // ========== 处理工具未找到 ==========
                    if !toolFound {
                        toolError = fmt.Errorf("tool '%s' not found", toolUse.Name)
                        fmt.Printf("\u001b[91merror\u001b[0m: %s\n", toolError.Error())
                    }
                    
                    // ========== 构造工具结果 ==========
                    if toolError != nil {
                        toolResults = append(toolResults, 
                            anthropic.NewToolResultBlock(toolUse.ID, toolError.Error(), true))
                        // 第三个参数 true 表示这是错误结果
                    } else {
                        toolResults = append(toolResults, 
                            anthropic.NewToolResultBlock(toolUse.ID, toolResult, false))
                    }
                }
            }
            
            // ========== 检查是否需要继续 ==========
            if !hasToolUse {
                break  // 没有工具调用，退出循环
            }
            
            // ========== 发送工具结果回 Claude ==========
            if a.verbose {
                log.Printf("Sending %d tool results back to Claude", len(toolResults))
            }
            
            toolResultMessage := anthropic.NewUserMessage(toolResults...)
            conversation = append(conversation, toolResultMessage)
            
            // ========== 再次调用 Claude ==========
            message, err = a.runInference(ctx, conversation)
            if err != nil {
                return err
            }
            conversation = append(conversation, message.ToParam())
            
            // 继续循环，处理可能的更多工具调用
        }
    }
}
```

```go
func Bash(input json.RawMessage) (string, error) {
    // ========== 1. 解析输入 ==========
    bashInput := BashInput{}
    err := json.Unmarshal(input, &bashInput)
    if err != nil {
        return "", err
    }
    
    log.Printf("Executing bash command: %s", bashInput.Command)
    
    // ========== 2. 构造命令 ==========
    cmd := exec.Command("bash", "-c", bashInput.Command)
    // exec.Command: 创建一个外部命令
    // 参数: bash -c "command"
    // -c: 从字符串读取命令
    
    // ========== 3. 执行命令 ==========
    output, err := cmd.CombinedOutput()
    // CombinedOutput: 运行命令并返回 stdout + stderr 的组合输出
    
    // ========== 4. 处理结果 ==========
    if err != nil {
        log.Printf("Bash command failed: %s, error: %v", bashInput.Command, err)
        
        // 关键设计：即使失败也返回输出，而不是 error
        // 这样 Claude 可以看到错误信息并尝试修复
        return fmt.Sprintf("Command failed with error: %s\nOutput: %s", 
            err.Error(), string(output)), nil
    }
    
    log.Printf("Bash command succeeded: %s (output: %d bytes)", 
        bashInput.Command, len(output))
    
    // ========== 5. 返回结果 ==========
    return strings.TrimSpace(string(output)), nil
    // TrimSpace: 去除首尾空白
}
```

```go
func EditFile(input json.RawMessage) (string, error) {
    // ========== 1. 解析输入 ==========
    editFileInput := EditFileInput{}
    err := json.Unmarshal(input, &editFileInput)
    if err != nil {
        return "", err
    }
    
    // ========== 2. 参数验证 ==========
    if editFileInput.Path == "" || editFileInput.OldStr == editFileInput.NewStr {
        log.Printf("EditFile failed: invalid input parameters")
        return "", fmt.Errorf("invalid input parameters")
    }
    
    log.Printf("Editing file: %s (replacing %d chars with %d chars)", 
        editFileInput.Path, len(editFileInput.OldStr), len(editFileInput.NewStr))
    
    // ========== 3. 读取文件 ==========
    content, err := os.ReadFile(editFileInput.Path)
    if err != nil {
        // ========== 文件不存在的情况 ==========
        if os.IsNotExist(err) && editFileInput.OldStr == "" {
            log.Printf("File does not exist, creating new file: %s", editFileInput.Path)
            return createNewFile(editFileInput.Path, editFileInput.NewStr)
        }
        log.Printf("Failed to read file %s: %v", editFileInput.Path, err)
        return "", err
    }
    
    oldContent := string(content)
    
    // ========== 4. 编辑模式选择 ==========
    var newContent string
    
    if editFileInput.OldStr == "" {
        // ========== 追加模式 ==========
        newContent = oldContent + editFileInput.NewStr
    } else {
        // ========== 替换模式 ==========
        
        // 4.1 检查匹配次数
        count := strings.Count(oldContent, editFileInput.OldStr)
        
        if count == 0 {
            log.Printf("EditFile failed: old_str not found in file %s", editFileInput.Path)
            return "", fmt.Errorf("old_str not found in file")
        }
        
        if count > 1 {
            log.Printf("EditFile failed: old_str found %d times in file %s, must be unique", 
                count, editFileInput.Path)
            return "", fmt.Errorf("old_str found %d times in file, must be unique", count)
        }
        
        // 4.2 执行替换
        newContent = strings.Replace(oldContent, 
            editFileInput.OldStr, editFileInput.NewStr, 1)
        // 最后一个参数 1: 只替换第一个匹配
    }
    
    // ========== 5. 写入文件 ==========
    err = os.WriteFile(editFileInput.Path, []byte(newContent), 0644)
    // 0644: 文件权限
    // - 6 (110): 所有者读写
    // - 4 (100): 组用户只读
    // - 4 (100): 其他用户只读
    
    if err != nil {
        log.Printf("Failed to write file %s: %v", editFileInput.Path, err)
        return "", err
    }
    
    log.Printf("Successfully edited file %s", editFileInput.Path)
    return "OK", nil
}
```

#### createNewFile 辅助函数

```go
func createNewFile(filePath, content string) (string, error) {
    log.Printf("Creating new file: %s (%d bytes)", filePath, len(content))
    
    // ========== 1. 创建目录 ==========
    dir := path.Dir(filePath)
    if dir != "." {
        log.Printf("Creating directory: %s", dir)
        err := os.MkdirAll(dir, 0755)
        // MkdirAll: 创建所有必要的父目录
        // 0755: 目录权限
        // - 7 (111): 所有者读写执行
        // - 5 (101): 组用户读执行
        // - 5 (101): 其他用户读执行
        
        if err != nil {
            log.Printf("Failed to create directory %s: %v", dir, err)
            return "", fmt.Errorf("failed to create directory: %w", err)
            // %w: 包装错误，保留原始错误信息
        }
    }
    
    // ========== 2. 写入文件 ==========
    err := os.WriteFile(filePath, []byte(content), 0644)
    if err != nil {
        log.Printf("Failed to create file %s: %v", filePath, err)
        return "", fmt.Errorf("failed to create file: %w", err)
```

```go
func CodeSearch(input json.RawMessage) (string, error) {
    // ========== 1. 解析输入 ==========
    codeSearchInput := CodeSearchInput{}
    err := json.Unmarshal(input, &codeSearchInput)
    if err != nil {
        return "", err
    }
    
    // ========== 2. 验证参数 ==========
    if codeSearchInput.Pattern == "" {
        log.Printf("CodeSearch failed: pattern is required")
        return "", fmt.Errorf("pattern is required")
    }
    
    log.Printf("Searching for pattern: %s", codeSearchInput.Pattern)
    
    // ========== 3. 构造 ripgrep 命令 ==========
    args := []string{"rg", "--line-number", "--with-filename", "--color=never"}
    // --line-number: 显示行号
    // --with-filename: 显示文件名
    // --color=never: 禁用颜色（方便解析）
    
    // ========== 4. 添加可选参数 ==========
    if !codeSearchInput.CaseSensitive {
        args = append(args, "--ignore-case")
    }
    
    if codeSearchInput.FileType != "" {
        args = append(args, "--type", codeSearchInput.FileType)
        // ripgrep 内置类型: go, js, py, rust, java, etc.
    }
    
    // ========== 5. 添加搜索模式 ==========
    args = append(args, codeSearchInput.Pattern)
    
    // ========== 6. 添加搜索路径 ==========
    if codeSearchInput.Path != "" {
        args = append(args, codeSearchInput.Path)
    } else {
        args = append(args, ".")  // 当前目录
    }
    
    // ========== 7. 执行命令 ==========
    cmd := exec.Command(args[0], args[1:]...)
    output, err := cmd.Output()
    
    // ========== 8. 处理结果 ==========
    if err != nil {
        if exitError, ok := err.(*exec.ExitError); ok && exitError.ExitCode() == 1 {
            // ripgrep 返回 exit code 1 表示无匹配（不是真正的错误）
            log.Printf("No matches found for pattern: %s", codeSearchInput.Pattern)
            return "No matches found", nil
        }
        log.Printf("Ripgrep command failed: %v", err)
        return "", fmt.Errorf("search failed: %w", err)
    }
    
    // ========== 9. 格式化输出 ==========
    result := strings.TrimSpace(string(output))
    lines := strings.Split(result, "\n")
    
    log.Printf("Found %d matches for pattern: %s", len(lines), codeSearchInput.Pattern)
    
    // ========== 10. 限制输出长度 ==========
    if len(lines) > 50 {
        result = strings.Join(lines[:50], "\n") + 
            fmt.Sprintf("\n... (showing first 50 of %d matches)", len(lines))
    }
    // 防止返回过多结果，影响响应速度
    
    return result, nil
}
```

```go
// 对话历史会无限增长，可能导致:
// 1. 内存耗尽
// 2. API 调用超过 token 限制

// 解决方案: 上下文窗口管理

func (a *Agent) trimConversation(conversation []anthropic.MessageParam) []anthropic.MessageParam {
    // 1. 计算当前 token 数
    tokens := countTokens(conversation)
    
    // 2. 如果超过限制，删除旧消息
    maxTokens := 100000  // Claude 的上下文限制
    if tokens > maxTokens {
        // 保留系统消息和最近的几轮对话
        keep := 10  // 保留最近 10 轮
        if len(conversation) > keep*2 {
            conversation = conversation[len(conversation)-keep*2:]
        }
    }
    
    return conversation
}
```

```go
func (a *Agent) executeToolsParallel(toolCalls []ToolCall) []ToolResult {
    results := make([]ToolResult, len(toolCalls))
    
    var wg sync.WaitGroup
    for i, call := range toolCalls {
        wg.Add(1)
        go func(idx int, call ToolCall) {
            defer wg.Done()
            result, _ := a.executeTool(call)
            results[idx] = result
        }(i, call)
    }
    wg.Wait()
    
    return results
```
